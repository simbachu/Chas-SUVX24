166def open_file(event=False):
167 global currentFilePath
168 file = filedialog.askopenfilename(filetypes = app.ftypes)
169 if file == '' or file == ():
170 return
171 window.title(app.name + " - " + str(file))
172 currentFilePath = file
173 try:
174 with open(file, 'r') as f:
175 txt.delete(1.0,END)
176 txt.insert(INSERT,f.read())
177 undo = [txt.get(1.0,END)]
178 except Exception as e:

180 messagebox.showerror("Open File Error", f"Could not open file:\n{file}\n\nError: {e}")
